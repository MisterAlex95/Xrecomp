cmake_minimum_required(VERSION 3.20)
project(host_perf_line_test C)
set(CMAKE_C_STANDARD 11)
if(WIN32)
    message(FATAL_ERROR "host_perf_line is POSIX-only (pthreads)")
endif()
# host_perf_line.c alone (XBA-243): the performance line's accounting and text, driven with made-up clocks.
find_package(Threads REQUIRED)
add_executable(host_perf_line_test test_main.c ../../src/platform/host_perf_line.c)
target_include_directories(host_perf_line_test PRIVATE ../../src/platform)
target_compile_options(host_perf_line_test PRIVATE -Wall -Wextra)
target_link_libraries(host_perf_line_test PRIVATE Threads::Threads m)
enable_testing()
add_test(NAME host_perf_line COMMAND host_perf_line_test)
