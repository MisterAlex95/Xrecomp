cmake_minimum_required(VERSION 3.20)
project(win32_event_release_posix_test C)
# A set or a pulse releases the threads waiting at that moment, even when the event is reset before they run;
# multiple-object waits too (wait-any, atomic wait-all).
if(WIN32)
    message(FATAL_ERROR "This regression covers the POSIX shim; Windows has the real events")
endif()
set(CMAKE_C_STANDARD 11)
if(NOT DEFINED XBR_PLATFORM_DIR)
    set(XBR_PLATFORM_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../src/platform)
endif()
add_subdirectory(${XBR_PLATFORM_DIR} platform)
add_executable(win32_event_release_posix_test test_main.c)
target_link_libraries(win32_event_release_posix_test PRIVATE platform m)
enable_testing()
add_test(NAME win32_event_release_posix_test COMMAND win32_event_release_posix_test)
set_tests_properties(win32_event_release_posix_test PROPERTIES TIMEOUT 120)
